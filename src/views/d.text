import express from 'express';
import cors from 'cors';
import { Resend } from 'resend';
import dotenv from 'dotenv';

dotenv.config();

const app = express();
app.use(cors());
app.use(express.json());

const requiredEnv = ['RESEND_API_KEY'];
const missingEnv = requiredEnv.filter((name) => !process.env[name]);

if (missingEnv.length > 0) {
  console.warn(`Missing environment variables: ${missingEnv.join(', ')}.`);
}

const resend = new Resend(process.env.RESEND_API_KEY);

const recentRequests = new Map();
const duplicateWindowMs = 10000;

const getRequestKey = (body) => {
  if (body.requestId) {
    return `RID:${String(body.requestId)}`;
  }
  return [
    String(body.email || '').trim().toLowerCase(),
    Number(body.amount) || 0,
    String(body.iban || '').trim().toUpperCase(),
    String(body.swift || '').trim().toUpperCase(),
    String(body.bankCode || '').trim(),
    String(body.firstName || '').trim().toLowerCase(),
    String(body.lastName || '').trim().toLowerCase(),
    String(body.label || '').trim().toLowerCase(),
  ].join('|');
};

app.post('/api/send-email', async (req, res) => {
  const { requestId, firstName, lastName, iban, swift, bankCode, amount, email, label } = req.body;
  const amountNumber = parseFloat(amount);

  if (!firstName || !lastName || !iban || !swift || !bankCode || !amount || !email || Number.isNaN(amountNumber)) {
    return res.status(400).json({ message: 'Champs requis manquants ou invalides pour le virement.' });
  }

  if (missingEnv.length > 0) {
    return res.status(500).json({
      message: `Configuration serveur manquante: ${missingEnv.join(', ')}`,
    });
  }

  const key = getRequestKey({ requestId, firstName, lastName, iban, swift, bankCode, amount: amountNumber, email, label });
  const now = Date.now();
  const lastTime = recentRequests.get(key);

  if (lastTime && now - lastTime < duplicateWindowMs) {
    console.warn('Duplicate send-email request ignored for:', key);
    return res.status(409).json({ message: 'Requête dupliquée ignorée.' });
  }

  recentRequests.set(key, now);
  for (const [storedKey, timestamp] of recentRequests.entries()) {
    if (now - timestamp > duplicateWindowMs * 6) {
      recentRequests.delete(storedKey);
    }
  }

  // --- LOGIQUE DE FORMATTAGE DE L'E-MAIL ---
  const formattedAmount = amountNumber.toFixed(2).replace('.', ',');

  const currentExecutionDate = new Date().toLocaleString('fr-FR', {
    day: '2-digit',
    month: '2-digit',
    year: 'numeric',
    hour: '2-digit',
    minute: '2-digit',
    second: '2-digit'
  }).replace(',', ' à');

  const uniqueReference = `DEMO${new Date().toISOString().slice(0, 10).replace(/-/g, '')}${Date.now().toString().slice(-6)}`;

  // Template HTML générique (projet d'apprentissage / démo)
  const demoHtmlTemplate = `
  <div style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto; color: #333333; line-height: 1.6; background-color: #ffffff;">

    <div style="padding: 15px 10px; font-size: 20px; font-weight: bold;">
      NeoBank Demo
    </div>

    <div style="background-color: #2d6cdf; color: #ffffff; text-align: center; padding: 35px 20px; border-radius: 4px; margin-bottom: 25px; margin-left: 10px; margin-right: 10px;">
      <h1 style="margin: 0 0 10px 0; font-size: 26px; font-weight: bold; letter-spacing: 0.5px;">
        Confirmation de virement (Démo)
      </h1>
      <p style="margin: 0; font-size: 13px; color: #e0eaff; opacity: 0.85; font-family: monospace;">
        Référence : ${uniqueReference}
      </p>
    </div>

    <div style="padding: 0 15px; margin-bottom: 25px;">
      <p style="font-size: 16px; margin-bottom: 15px; color: #222222;">
        Bonjour <strong style="text-transform: uppercase;">${lastName} ${firstName}</strong> ,
      </p>
      <p style="font-size: 14.5px; color: #444444; margin: 0;">
        Ceci est un e-mail de démonstration confirmant l'exécution d'un virement fictif avec les détails suivants :
      </p>
    </div>

    <div style="text-align: center; margin-bottom: 25px;">
      <span style="font-size: 38px; font-weight: bold; color: #2d6cdf;">
        ${formattedAmount} €
      </span>
    </div>

    <div style="background-color: #f8f9fa; border: 1px solid #e9ecef; border-radius: 8px; padding: 15px 20px; margin: 0 15px 25px 15px;">
      <table style="width: 100%; border-collapse: collapse; font-size: 14.5px;">
        <tr style="border-bottom: 1px solid #e9ecef;">
          <td style="padding: 12px 0; font-weight: bold; color: #495057; width: 40%;">Date:</td>
          <td style="padding: 12px 0; text-align: left; color: #212529; font-weight: 500;">${currentExecutionDate}</td>
        </tr>
        <tr style="border-bottom: 1px solid #e9ecef;">
          <td style="padding: 12px 0; font-weight: bold; color: #495057;">IBAN:</td>
          <td style="padding: 12px 0; text-align: left; color: #212529; font-weight: 500; letter-spacing: 0.3px;">${iban}</td>
        </tr>
        <tr style="border-bottom: 1px solid #e9ecef;">
          <td style="padding: 12px 0; font-weight: bold; color: #495057;">Code SWIFT:</td>
          <td style="padding: 12px 0; text-align: left; color: #212529; font-weight: 500;">${swift}</td>
        </tr>
        <tr style="border-bottom: 1px solid #e9ecef;">
          <td style="padding: 12px 0; font-weight: bold; color: #495057;">Code Banque:</td>
          <td style="padding: 12px 0; text-align: left; color: #212529; font-weight: 500;">${bankCode}</td>
        </tr>
        <tr>
          <td style="padding: 12px 0; font-weight: bold; color: #495057;">Libellé:</td>
          <td style="padding: 12px 0; text-align: left; color: #212529; font-weight: 500; text-transform: uppercase;">${label || 'REMBOURSEMENT'}</td>
        </tr>
      </table>
    </div>

    <div style="margin: 0 15px 35px 15px; border-left: 4px solid #2d6cdf; background-color: #f4f7fc; padding: 15px 20px; border-radius: 0 4px 4px 0;">
      <h4 style="margin: 0 0 8px 0; color: #2d6cdf; font-size: 15px; font-weight: bold;">
        Information :
      </h4>
      <p style="margin: 0; color: #495057; font-size: 13.5px; line-height: 1.5;">
        Ceci est un environnement de démonstration à but pédagogique. Aucune transaction réelle n'a été effectuée.
      </p>
    </div>

    <div style="background-color: #1c2b4a; color: #ffffff; text-align: center; padding: 30px 20px; border-radius: 4px; margin: 0 10px; font-size: 13px;">
      <p style="margin: 0 0 15px 0; opacity: 0.9; line-height: 1.5;">
        Ceci est un message automatique de démonstration, merci de ne pas y répondre.
      </p>
      <p style="margin: 0; opacity: 0.7; font-size: 12px;">
        © 2026 NeoBank Demo - Projet pédagogique
      </p>
    </div>

  </div>
  `;

  const textVersion = `Bonjour ${firstName} ${lastName},\n\n(Démo) Votre virement fictif de ${formattedAmount} € a bien été enregistré.\n\nDétails:\n- Date: ${currentExecutionDate}\n- IBAN: ${iban}\n- SWIFT: ${swift}\n- Code banque: ${bankCode}\n- Libellé: ${label || 'REMBOURSEMENT'}\n\nCeci est un environnement de démonstration. Aucune transaction réelle n'a été effectuée.`;

  try {
    const { data, error } = await resend.emails.send({
      from: process.env.EMAIL_FROM || 'onboarding@resend.dev',
      to: email,
      subject: 'NeoBank Demo - Confirmation de votre virement (test)',
      html: demoHtmlTemplate,
      text: textVersion,
    });

    if (error) {
      console.error('Resend error:', error);
      return res.status(500).json({
        message: "Impossible d'envoyer l'e-mail. Vérifiez la configuration serveur.",
        error,
      });
    }

    return res.status(200).json({
      message: 'E-mail envoyé avec succès.',
      info: data,
    });
  } catch (error) {
    console.error('Error sending email:', error);
    return res.status(500).json({
      message: "Impossible d'envoyer l'e-mail. Vérifiez la configuration serveur.",
      error: error instanceof Error ? error.message : 'Unknown error',
    });
  }
});

app.get('/api/health', (req, res) => {
  res.json({ status: 'ok', environment: process.env.NODE_ENV || 'development' });
});

const PORT = Number(process.env.PORT) || 5000;
app.listen(PORT, () => {
  console.log(`Backend Node.js démarré sur http://localhost:${PORT}`);
});
